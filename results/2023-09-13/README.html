\(F_{ST}\)s among fish species/ecotypes

Mar Llaberia-Robledillo & J. Ignacio Lucas Lledó

2024-01-03

\(F_{ST}\) values must be estimated among populations, which are defined not only by the species assigned but by the lake as well. Members of the same species living in different lakes are from different populations, and we should separate them when estimating \(F_{ST}\)s.

Some combinations of species and lake are not represented by many individuals. Only populations with at least 5 individuals sampled may be used for \(F_{ST}\) estimation. Populations based on ambiguous species assignation (e.g. Bri.C.brienzii/C.fatioi, with 5 individuals) may be ignored.

FISHDATA=../../data/Fish_clean2.tsv
VCF=/gata/mar/cophyhopa/results/2022-04-22/assem2_outfiles/assem2.vcf
if [ ! -e ourFish.txt ]; then
   gawk '(/^#CHROM/){
      for (i = 10; i <= NF; i++) print $i
      exit
   }' $VCF > ourFish.txt
fi

if [ ! -e ourPopulations.txt ]; then
   grep -f ourFish.txt $FISHDATA | cut -f 2,19 | \
   gawk '{
      POP = substr($1, 1, 3) "." $2
      F[POP]++l
   }END{
      for (f in F) {
         if (F[f] >= 5) print f
      }
   }' | \
   grep -v "/" | sort > ourPopulations.txt
fi

for POP in $(cat ourPopulations.txt); do
   if [ ! -e ${POP}.txt ]; then
      gawk -v POP=${POP} '{
         split(POP,A,/\./)
         if (($2 ~ A[1]) && ($19 ~ A[3])) print $1
      }' $FISHDATA > ${POP}.txt
   fi
done

if [ ! -d FST ]; then mkdir FST; fi

if [ ! -e dirty.recode.vcf ]; then
  vcftools --vcf $VCF \
           --out dirty \
           --remove ../2022-05-04/TheWorst.txt \
           --recode \
           --recode-INFO-all > vcftools.log 2>&1
fi
#ALPINE=(acrinasus alpinus confusus duplex fatioi profundus steinmanni)
#ARCTIC=(D L P)
ALL=($(cat ourPopulations.txt))
for j in $(seq 1 $(( ${#ALL[@]} - 1 )) ); do
   for i in $(seq 0 $(( j - 1 )) ); do
      if [ ! -e FST/${ALL[$i]}.${ALL[$j]}.weir.fst ]; then
         vcftools --vcf dirty.recode.vcf \
                  --out FST/${ALL[$i]}.${ALL[$j]} \
                  --thin 500 \
                  --remove-indels \
                  --mac 2 \
                  --max-alleles 2 \
                  --max-missing 1 \
                  --max-meanDP 1000 \
                  --weir-fst-pop ${ALL[$i]}.txt \
                  --weir-fst-pop ${ALL[$j]}.txt >> vcftools.log 2>&1 &
      fi
   done
done
wait

if [ ! -e summary.txt ]; then
   echo -e "Pop1\tPop2\tMean.FST" > summary.txt
   for j in $(seq 1 $(( ${#ALL[@]} - 1 )) ); do
      for i in $(seq 0 $(( j - 1 )) ); do
         gawk -v P1=${ALL[$i]} -v P2=${ALL[$j]} '(NR > 1){
            S += $3
         }END{
            MEANFST = S / (NR - 1)
            print P1 "\t" P2 "\t" MEANFST
         }' FST/${ALL[$i]}.${ALL[$j]}.weir.fst >> summary.txt
      done
   done
fi
FSTpop <- read.table('summary.txt', header = TRUE)
Pops   <- unique(c(FSTpop$Pop1, FSTpop$Pop2))
FSTmat <- matrix(0, nrow = length(Pops), ncol = length(Pops))
row.names(FSTmat) <- Pops
colnames(FSTmat)  <- Pops
for (i in 1:dim(FSTpop)[1]) {
   FSTmat[FSTpop[i, 1], FSTpop[i, 2]] <- FSTpop[i, 3]
   FSTmat[FSTpop[i, 2], FSTpop[i, 1]] <- FSTpop[i, 3]
}
heatmap(FSTmat, scale = 'none', margins = c(6, 5))
